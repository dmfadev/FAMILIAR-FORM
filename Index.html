// ── GET → devuelve datos iniciales al formulario ────────────── function doGet(e) { try { return respuestaJSON_(getDatosIniciales_()); } catch(err) { return respuestaJSON_({ ok: false, error: err.message }); } } // ── POST → recibe y guarda el registro ─────────────────────── function doPost(e) { try { const payload = JSON.parse(e.postData.contents); return respuestaJSON_(guardarRegistro_(payload)); } catch(err) { return respuestaJSON_({ ok: false, error: err.message }); } } function respuestaJSON_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ── DATOS INICIALES ─────────────────────────────────────────── function getDatosIniciales_() { const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); const hSuc = ss.getSheetByName(CONFIG.SHEET_SUCURSALES); const hMod = ss.getSheetByName(CONFIG.SHEET_MODELO); // LOCALES: col A = Tienda, col B = Shopping const rowsSuc = Math.max(0, hSuc.getLastRow() - 1); const locales = rowsSuc > 0 ? hSuc.getRange(2, 1, rowsSuc, 2).getValues() .map(r => { const t = String(r[0]).trim(); const s = String(r[1]).trim(); if (!t && !s) return ''; if (!s) return t; if (!t) return s; return t + ' — ' + s; }).filter(Boolean) : []; // MODELO: cada columna es una lista independiente // Col A: Fiscalizadores // Col B: Tiene materiales (SI/NO — no se usa como lista) // Col C: Beneficios // Col D: Tipos de Oferta // Col E: Tipo de Material (A/B — no se usa como lista) // Col F: Materiales Tipo B // Col G: Materiales Tipo A // Col H: Estado Visual function col(n) { const total = hMod.getLastRow() - 1; if (total <= 0) return []; return hMod.getRange(2, n, total, 1).getValues() .map(r => String(r[0]).trim()) .filter(Boolean); } return { ok: true, locales, fiscalizadores: col(1), beneficios: col(3), tiposOferta: col(4), materialesB: col(6), materialesA: col(7), estadoVisual: col(8), }; } // ── GUARDAR REGISTRO ────────────────────────────────────────── function guardarRegistro_(payload) { if (!payload.fotos || payload.fotos.length < CONFIG.MIN_FOTOS) return { ok: false, error: 'Se requiere al menos ' + CONFIG.MIN_FOTOS + ' foto.' }; if (payload.fotos.length > CONFIG.MAX_FOTOS) return { ok: false, error: 'Maximo ' + CONFIG.MAX_FOTOS + ' fotos.' }; // Subir fotos a Drive const urls = []; for (let i = 0; i < payload.fotos.length; i++) { const r = guardarFoto_(payload.fotos[i].base64, payload.fotos[i].nombre); if (!r.ok) return { ok: false, error: 'Error foto ' + (i+1) + ': ' + r.error }; urls.push(r.url); } // Fecha y hora del servidor (anti-fraude) const tz = Session.getScriptTimeZone(); const ahora = new Date(); const fecha = Utilities.formatDate(ahora, tz, 'dd/MM/yyyy'); const hora = Utilities.formatDate(ahora, tz, 'HH:mm:ss'); const matsA = Array.isArray(payload.materialesA) ? payload.materialesA.join(', ') : ''; const matsB = Array.isArray(payload.materialesB) ? payload.materialesB.join(', ') : ''; const fila = [ fecha, // FECHA hora, // HORA payload.fiscalizador || '', // FISCALIZADOR fecha, // FECHA DE LA ENCUESTA (server-side) payload.local || '', // LOCALES payload.tieneMateriales || '', // CUENTA CON MATERIALES payload.beneficio || '', // BENEFICIO payload.tipoOferta || '', // TIPO OFERTA payload.tipoMaterial || '', // TIPO MATERIAL matsB, // MATERIALES TIPO B matsA, // MATERIALES TIPO A payload.estadoVisual || '', // ESTADO VISUAL ...Array.from({length: CONFIG.MAX_FOTOS}, (_, i) => urls[i] || ''), payload.observaciones || '', // OBSERVACIONES ]; // Obtener o crear hoja RESPUESTAS const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); let sheet = ss.getSheetByName(CONFIG.SHEET_RESPUESTAS); if (!sheet) { sheet = ss.insertSheet(CONFIG.SHEET_RESPUESTAS); const headers = [ 'FECHA', 'HORA', 'FISCALIZADOR', 'FECHA DE LA ENCUESTA', 'LOCALES', '¿EL LOCAL CUENTA CON LOS MATERIALES DESIGNADOS?', '¿CUÁL ES EL BENEFICIO QUE OFRECE ESTE ALIADO?', '¿QUÉ TIPO DE OFERTA CORRESPONDE A ESTE LOCAL?', '¿QUÉ TIPO DE MATERIAL TIENE ESTE ALIADO ASIGNADO?', '[SI ES TIPO B] ¿QUÉ MATERIALES TIPO B ESTÁN COLOCADOS?', '[SI ES TIPO A] ¿QUÉ MATERIALES TIPO A ESTÁN COLOCADOS?', 'EL MATERIAL ESTÁ EN BUEN ESTADO Y VISIBLE PARA EL CLIENTE?', 'FOTO 1','FOTO 2','FOTO 3','FOTO 4','FOTO 5','FOTO 6', 'OBSERVACIONES', ]; sheet.getRange(1, 1, 1, headers.length) .setValues([headers]) .setFontWeight('bold') .setBackground('#003087') .setFontColor('#ffffff'); sheet.setFrozenRows(1); } const row = sheet.getLastRow() + 1; sheet.getRange(row, 1, 1, fila.length).setValues([fila]); // Insertar =IMAGE() en columnas de fotos (columnas 13 a 18) for (let i = 0; i < urls.length; i++) { sheet.getRange(row, 13 + i) .setFormula('=IMAGE("' + urls[i] + '",4,80,80)'); } if (urls.length) sheet.setRowHeight(row, 90); return { ok: true, fila: row, fotos: urls }; } // ── SUBIR FOTO A DRIVE ──────────────────────────────────────── function guardarFoto_(base64Data, nombre) { try { const raw = base64Data.replace(/^data:image\/\w+;base64,/, ''); const blob = Utilities.newBlob( Utilities.base64Decode(raw), 'image/jpeg', nombre ); const file = DriveApp.getFolderById(CONFIG.DRIVE_FOLDER_ID).createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return { ok: true, url: 'https://drive.google.com/uc?export=view&id=' + file.getId(), }; } catch(e) { return { ok: false, error: e.message }; } }
